    UOM = "CAJ",

    // Mes anterior cerrado -> formato IBP: "yy-MMM" (ej: 26-Jan)

    DiaCierre = 5,
    MesesAtras = if Date.Day(Hoy) > DiaCierre then -1 else -2,

                #"$format" = "json"
            ],
            Headers = [Accept = "application/json"]
        ]
    ),

    Json = Json.Document(Response),

    // OData v2 (d/results) y v4 (value)
    Data =
        if Record.HasFields(Json, "d") and Record.HasFields(Json[d], "results") then
            Json[d][results]
        else if Record.HasFields(Json, "value") then
            Json[value]
        else
            {},

    Tabla = Table.FromRecords(if Data is list then Data else {}),

    TablaSinMeta =
        if List.Contains(Table.ColumnNames(Tabla), "__metadata") then
            Table.RemoveColumns(Tabla, {"__metadata"})
        else
            Tabla,

    // =========================
    // Renombres “business friendly” (opcional)
    // =========================
    ProductosRenombrados =
        Table.RenameColumns(
            TablaSinMeta,
            {
                {"PRDID", "Product ID"},
                {"UOMTOID", "UOM"},
                {"ZBIGDIVISION", "Gran División"},
                {"ZBIGBUSINESS", "Gran Negocio"},

- IBP - Historico Entrega

let
    // =====================================================
    // ✅ PARÁMETRO: cantidad de meses históricos cerrados
    //    Ej: 3 => trae Oct, Nov, Dec si hoy es Enero
    // =====================================================
    nMeses = 13,

    // =====================================================
    // 🌐 Base URL + Planning Area
    // =====================================================
    BaseUrl = "https://my308646-api.scmibp1.ondemand.com",
    PlanningArea = "MOLIBP",
    RelativePath = "sap/opu/odata/IBP/EXTRACT_ODATA_SRV/" & PlanningArea,

    // =====================================================
    // 📅 Lista de meses cerrados: últimos nMeses (incluye mes anterior)
    // =====================================================
    hoy = Date.From(DateTime.LocalNow()),
    fechaMesAnterior = Date.AddMonths(Date.StartOfMonth(hoy), -1),          // 1° día del mes anterior
    fechasMeses = List.Transform({0..nMeses-1}, each Date.AddMonths(fechaMesAnterior, -_)),

    // =====================================================
    // 🔤 Funciones auxiliares
    // =====================================================
    MonthAbbrev_EN = (d as date) as text =>
        Text.Start(Date.ToText(d, "MMM", "en-US"), 1) &
        Text.Lower(Text.Range(Date.ToText(d, "MMM", "en-US"), 1)),           // Jan, Feb, ...

    Periodo_YYMMM = (d as date) as text =>
        Text.End(Text.From(Date.Year(d)), 2) & "-" & MonthAbbrev_EN(d),      // 26-Jan

    // =====================================================
    // ⚙️ Función: bajar datos por período (yy-MMM)
    // =====================================================
    ObtenerDatosPorPeriodo = (periodo as text) as table =>
        let
            Response =
                Web.Contents(
                    BaseUrl,
                    [
                        RelativePath = RelativePath,
                        Query = [
                            #"$select" = "PERIODID3,LOCID,PRDID,ZAREAGC,ACTUALSQTY,UOMTOID,ZVIGENCIA,ZAREAFCST",
                            #"$filter" =
                                "PERIODID3 eq '" & periodo & "' " &
                                "and UOMTOID eq 'UMG' " &
                                "and LOCID eq 'AR01' " &
                                "and ZVIGENCIA eq '1' " &
                                "and ZAREAFCST ne 'Comex' " &
                                "and ZAREAFCST ne 'Tienda Molinos' " &
                                "and ZCLIENTEBOCA eq 'Cliente'",
                            #"$format" = "json"
                        ],
                        Headers = [Accept = "application/json"]
                    ]
                ),
            Json  = Json.Document(Response),
            Data  = try Json[d][results] otherwise {},
            Tabla = Table.FromRecords(Data)
        in
            Tabla,

    // =====================================================
    // 📥 Ejecutar para cada mes y combinar
    // =====================================================
    periodos = List.Transform(fechasMeses, each Periodo_YYMMM(_)),
    tablas   = List.Transform(periodos, each ObtenerDatosPorPeriodo(_)),
    DatosCombinados = Table.Combine(tablas),

    // =====================================================
    // 🧹 Quitar columnas si existen (robusto)
    // =====================================================
    colsObjetivo  = {"__metadata","UOMTOID","ZVIGENCIA","ZAREAFCST","LOCID"},
    colsPresentes = List.Intersect({Table.ColumnNames(DatosCombinados), colsObjetivo}),
    OrigenSinColumnas =
        if List.Count(colsPresentes) > 0
        then Table.RemoveColumns(DatosCombinados, colsPresentes)
        else DatosCombinados,

    // =====================================================
    // ✏️ Renombrar (ignora si falta alguna)
    // =====================================================
    OrigenRenombrado = Table.RenameColumns(OrigenSinColumnas, {
        {"PERIODID3", "Time Periods"},
        {"PRDID", "Product ID"},
        {"ZAREAGC", "Area GC"},
        {"ACTUALSQTY", "Histórico de Ventas"}
    }, MissingField.Ignore),

    // =====================================================
    // 📆 'yy-MMM' → date (1° de mes)
    // =====================================================
    AddFirstDayOfMonth = Table.AddColumn(OrigenRenombrado, "FirstDayOfMonth", each
        let
            parts = Text.Split([Time Periods], "-"),
            y = if List.Count(parts)=2 then "20" & Text.Start(parts{0}, 2) else null,
            m = if List.Count(parts)=2 then
                    1 + List.PositionOf(
                        {"JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"},
                        Text.Upper(parts{1})
                    )
                else null
        in
            if y <> null and m <> null and m > 0
            then try Date.FromText(y & "-" & Text.PadStart(Text.From(m), 2, "0") & "-01") otherwise null
            else null,
        type date
    ),

    ConFecha = Table.RenameColumns(
        Table.RemoveColumns(AddFirstDayOfMonth, {"Time Periods"}),
        {{"FirstDayOfMonth", "Time Periods"}}
    ),

    // =====================================================
    // 🔢 Limpieza numérica robusta ("," miles; "." decimal)
    // =====================================================
    #"Valor a texto"          = Table.TransformColumnTypes(ConFecha, {{"Histórico de Ventas", type text}}),
    #"Sin separador de miles" = Table.TransformColumns(#"Valor a texto", {{"Histórico de Ventas", each if _ = null then null else Text.Replace(_, ",", ""), type text}}),
    #"Valor a número"         = Table.TransformColumns(#"Sin separador de miles", {{"Histórico de Ventas", each if _ = null or _ = "" then null else Number.FromText(_, "en-US"), type number}}),

    // =====================================================
    // 📊 Orden final
    // =====================================================
    ColumnasReordenadas = Table.ReorderColumns(#"Valor a número",
        {"Time Periods","Product ID","Area GC","Histórico de Ventas"})
in
    ColumnasReordenadas

- IBP - Forecast

let
    // 🕒 Control de rango
    mesActual = Date.StartOfMonth(DateTime.LocalNow()),
    inicio    = Date.AddMonths(mesActual, -13), // 3 meses hacia atrás (excluye el actual)
    fin       = Date.AddMonths(mesActual,  2), // 2 meses hacia adelante (incluye el actual)

    // 🌐 Base URL
    BaseUrl = "https://my308646-api.scmibp1.ondemand.com",

    // 📅 Generar todos los meses del rango (1° de cada mes)
    FechasRango = List.Generate(
        () => inicio,
        (d) => d <= fin,
        (d) => Date.AddMonths(d, 1)
    ),

    // 🧭 Mapear fechas a records {Año, Mes="Jan"...}
    meses = {"Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"},
    AñoMesTotal = List.Transform(
        FechasRango,
        (d) => [Año = Date.Year(d), Mes = meses{Date.Month(d) - 1}]
    ),

    // ⚙️ Función por período yy-MMM
    ObtenerDatosPorAñoMes = (añoMes as record) as table =>
        let
            añoTexto = Text.End(Text.From(añoMes[Año]), 2),     // "2025" → "25"
            periodo  = añoTexto & "-" & añoMes[Mes],
            Response = Web.Contents(
                BaseUrl,
                [
                    RelativePath = "sap/opu/odata/IBP/EXTRACT_ODATA_SRV/MOLIBP",
                    Query = [
                        #"$select" = "PERIODID3,LOCID,PRDID,ZAREAGC,STATISTICALFORECASTQTY,DEMANDPLANNINGQTY,UOMTOID,ZVIGENCIA,ZAREAFCST",
                        #"$filter" =
                            "PERIODID3 eq '" & periodo & "' " &
                            "and UOMTOID eq 'UMG' " &
                            "and LOCID eq 'AR01' " &
                            "and ZVIGENCIA  eq '1' " &
                            "and ZAREAFCST ne 'Comex' " &
                            "and ZAREAFCST ne 'Tienda Molinos' " &
                            "and ZCLIENTEBOCA eq 'Cliente'",
                        #"$format" = "json"
                    ],
                    Headers = [Accept = "application/json"]
                ]
            ),
            Json  = Json.Document(Response),
            Data  = try Json[d][results] otherwise {},
            Tabla = Table.FromRecords(Data)
        in
            Tabla,

    // 📥 Ejecutar para todo el rango
    DatosPorMes     = List.Transform(AñoMesTotal, each ObtenerDatosPorAñoMes(_)),
    DatosCombinados = Table.Combine(DatosPorMes),

    // 🧹 Remover columnas solo si existen
    colsDrop  = {"__metadata","UOMTOID","ZVIGENCIA","ZAREAFCST","LOCID"},
    presentes = List.Intersect({Table.ColumnNames(DatosCombinados), colsDrop}),
    OrigenSinColumnas =
        if List.Count(presentes) > 0
        then Table.RemoveColumns(DatosCombinados, presentes)
        else DatosCombinados,

    // ✏️ Renombrar (ignora faltantes)
    OrigenRenombrado = Table.RenameColumns(OrigenSinColumnas, {
        {"PERIODID3", "Time Periods"},
        {"PRDID", "Product ID"},
        {"ZAREAGC", "Area GC"},
        {"STATISTICALFORECASTQTY", "Forecast Estadístico"},
        {"DEMANDPLANNINGQTY", "Forecast Consensuado"}
    }, MissingField.Ignore),

    // 📆 'yy-MMM' → date (1° de mes)
    AddFirstDayOfMonth = Table.AddColumn(OrigenRenombrado, "FirstDayOfMonth", each 
        let
            parts = Text.Split([Time Periods], "-"),
            dt =
                if List.Count(parts) = 2 then
                    let
                        y = "20" & Text.Start(parts{0}, 2),
                        m = 1 + List.PositionOf(
                                {"JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"},
                                Text.Upper(parts{1})
                            )
                    in
                        try Date.FromText(y & "-" & Text.PadStart(Text.From(m), 2, "0") & "-01") otherwise null
                else
                    null
        in
            dt,
        type date
    ),

    // 📇 Reemplazar columna original por fecha
    ConFecha = Table.RenameColumns(
        Table.RemoveColumns(AddFirstDayOfMonth, {"Time Periods"}),
        {{"FirstDayOfMonth","Time Periods"}}
    ),

    // 🔢 Conversión robusta a número (ambas columnas)
    ToNumber = (x as any) as nullable number =>
        let
            t = try Text.From(x) otherwise null,
            t2 = if t = null then null else Text.Replace(t, ",", ""),   // quita separador miles
            n = try Number.FromText(t2, "en-US") otherwise null
        in
            n,

    Convertidos = Table.TransformColumns(
        ConFecha,
        {
            {"Forecast Estadístico", each ToNumber(_), type number},
            {"Forecast Consensuado", each ToNumber(_), type number}
        },
        null,
        MissingField.Ignore
    ),

    // 📊 Orden final (acá va una LISTA, no una tabla)
    ColumnasReordenadas = Table.ReorderColumns(
        Convertidos,
        {"Time Periods","Product ID","Area GC","Forecast Estadístico","Forecast Consensuado"},
        MissingField.Ignore
    )
in
    ColumnasReordenadas

- IBP - Margen

let
    // =========================
    // Parámetros editables
    // =========================
    BaseUrl = "https://my308646-api.scmibp1.ondemand.com",
    RelativePath = "sap/opu/odata/IBP/EXTRACT_ODATA_SRV/MOLIBP",
    UOM = "CAJ",
    CURR = "USD",

    // =========================
    // Función que trae la tabla cruda para un período dado
    // =========================
    TraerPeriodo = (PeriodoYYMMM as text) as table =>
        let
            Response = Web.Contents(
                BaseUrl,
                [
                    RelativePath = RelativePath,
                    Query = [
                        #"$select" = "PRDID,UOMTOID,CURRTOID,ZMARGENUNITARIOUSD",
                        #"$filter" = "UOMTOID eq '" & UOM & "' and PERIODID3 eq '" & PeriodoYYMMM & "' and CURRTOID eq '" & CURR & "'",
                        #"$format" = "json"
                    ],
                    Headers = [Accept = "application/json"]
                ]
            ),
            Json = Json.Document(Response),
            Data =
                if Record.HasFields(Json, "d") and Record.HasFields(Json[d], "results") then
                    Json[d][results]
                else if Record.HasFields(Json, "value") then
                    Json[value]
                else
                    {},
            Tabla = Table.FromRecords(if Data is list then Data else {}),
            TablaSinMeta =
                if List.Contains(Table.ColumnNames(Tabla), "__metadata") then
                    Table.RemoveColumns(Tabla, {"__metadata"})
                else
                    Tabla
        in
            TablaSinMeta,

    // =========================
    // Períodos: mes en curso y mes anterior (formato "yy-MMM")
    // =========================
    Hoy = Date.From(DateTime.LocalNow()),
    FmtPeriodo = (d as date) as text =>
        Text.End(Text.From(Date.Year(d)), 2) & "-" & Date.ToText(d, "MMM", "en-US"),

    PeriodoActual = FmtPeriodo(Date.StartOfMonth(Hoy)),
    PeriodoPrevio = FmtPeriodo(Date.AddMonths(Date.StartOfMonth(Hoy), -1)),

    // =========================
    // Fallback: mes en curso, si viene vacío usa el mes anterior
    // =========================
    TablaActual = TraerPeriodo(PeriodoActual),
    TablaElegida =
        if Table.RowCount(TablaActual) > 0 then
            TablaActual
        else
            TraerPeriodo(PeriodoPrevio),

    // =========================
    // Conversión numérica robusta (margen)
    // =========================
    ToNumber = (x as any) as nullable number =>
        let
            t = try Text.From(x) otherwise null,
            t2 = if t = null then null else Text.Replace(t, ",", ""),
            n = try Number.FromText(t2, "en-US") otherwise null
        in
            n,
    Convertidos =
        Table.TransformColumns(
            TablaElegida,
            {{"ZMARGENUNITARIOUSD", each ToNumber(_), type number}},
            null,
            MissingField.Ignore
        ),

    // =========================
    // Agregación a nivel SKU
    // =========================
    Agrupado =
        Table.Group(
            Convertidos,
            {"PRDID"},
            {{"Margen Unitario (USD)", each List.Max([ZMARGENUNITARIOUSD]), type number}}
        ),

    // =========================
    // Renombre final
    // =========================
    Renombrado =
        Table.RenameColumns(
            Agrupado,
            {{"PRDID", "SKU"}},
            MissingField.Ignore
        )
in
    Renombrado
