let
    // === CARGAR LAS DOS TABLAS BASE ===

    ZFCST = ZFCST,   // <-- Reemplazá con el nombre real de la consulta ZFCST

    // === PREPARAR HISTÓRICO ===
    HistoricoFiltrado = Table.SelectRows(ZHIST, each [Tipo de Historia] = "Historico Ajustado"),

        {"Tipo de Historia", "Tipo"},

    }),
    HistoricoEtiquetado = Table.AddColumn(HistoricoRenombrado, "Origen", each "Histórico", type text),

    ForecastEtiquetado = Table.AddColumn(ForecastRenombrado, "Origen", each "Forecast", type text),

- ZHIST

let
    // 📆 Lista de años y meses (editá libremente)
    años  = {2021, 2022, 2023, 2024, 2025,2026},
    meses = {"Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"},

    // 📅 Fechas: hasta el mes anterior
    fechaFin = Date.AddMonths(Date.StartOfMonth(DateTime.LocalNow()), -1),

    // 🌐 Base URL del servicio
    BaseUrl = "https://my308646-api.scmibp1.ondemand.com",

    // 🔁 Combinaciones año-mes
    AñoMes = List.TransformMany(años, each meses, (año, mes) => [Año = año, Mes = mes]),

    // 🧮 Mapear a fecha (1° de mes) y filtrar hasta mes anterior
    MesANum = {"Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"},
    AñoMesFiltrado =
        List.Select(
            AñoMes,
            (r) =>
                let
                    m = 1 + List.PositionOf(MesANum, r[Mes]),
                    f = Date.FromText(Text.From(r[Año]) & "-" & Text.PadStart(Text.From(m), 2, "0") & "-01")
                in
                    Date.From(f) <= Date.From(fechaFin)

        ),

    // ⚙️ Función por período yy-MMM (proper case)
    ObtenerDatosPorAñoMes = (añoMes as record) as table =>
        let
            periodo  = Text.End(Text.From(añoMes[Año]), 2) & "-" & añoMes[Mes],   // e.g. 25-Jul
            Response = Web.Contents(
                BaseUrl,
                [
                    RelativePath = "sap/opu/odata/IBP/EXTRACT_ODATA_SRV/MOLIBP",
                    Query = [
                        #"$select" = "PERIODID3,PRDID,ZAREAGC,ACTUALSQTY,ADJUSTEDACTUALSQTY,UOMTOID",
                        #"$filter" = "PERIODID3 eq '" & periodo & "' and UOMTOID eq 'UMG' and ZVIGENCIA eq '1' and ZAREAFCST ne 'Comex' and ZCLIENTEBOCA eq 'Cliente'",
                        #"$format" = "json"
                    ],
                    Headers = [Accept = "application/json"]
                ]
            ),
            Json  = Json.Document(Response),
            Data  = try Json[d][results] otherwise {},
            Tabla = Table.FromRecords(Data)
        in
            Tabla,

    // 📥 Ejecutar para cada combinación y unificar
    DatosPorMes       = List.Transform(AñoMesFiltrado, each ObtenerDatosPorAñoMes(_)),
    DatosCombinados   = Table.Combine(DatosPorMes),

    // 🧹 Quitar columnas si existen (robusto)
    colsObjetivo   = {"__metadata","UOMTOID"},
    colsPresentes  = List.Intersect({Table.ColumnNames(DatosCombinados), colsObjetivo}),
    OrigenSinColumnas =
        if List.Count(colsPresentes) > 0
        then Table.RemoveColumns(DatosCombinados, colsPresentes)
        else DatosCombinados,

    // ✏️ Renombrar (ignora si falta alguna)
    OrigenRenombrado = Table.RenameColumns(OrigenSinColumnas, {
        {"PERIODID3", "Time Periods"},
        {"PRDID", "Product ID"},
        {"ZAREAGC", "Area/GC"},
        {"ACTUALSQTY", "Histórico de Ventas"},
        {"ADJUSTEDACTUALSQTY", "Historico Ajustado"}
    }, MissingField.Ignore),

    // 📆 'yy-MMM' → date (1° de mes) usando mapa en MAYÚSCULAS
    AddFirstDayOfMonth = Table.AddColumn(OrigenRenombrado, "FirstDayOfMonth", each 
        let
            parts = Text.Split([Time Periods], "-"),
            dt =
                if List.Count(parts) = 2 then
                    let
                        y = "20" & Text.Start(parts{0}, 2),
                        m = 1 + List.PositionOf({"JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"}, Text.Upper(parts{1}))
                    in
                        try Date.FromText(y & "-" & Text.PadStart(Text.From(m), 2, "0") & "-01") otherwise null
                else
                    null
        in
            dt,
        type date
    ),

    // 🏷️ Reemplazar columna original por la fecha real
    ConFecha = Table.RenameColumns(
        Table.RemoveColumns(AddFirstDayOfMonth, {"Time Periods"}),
        {{"FirstDayOfMonth", "Time Periods"}}
    ),

    // 🔄 Unpivot a formato largo
    DatosDesplegados = Table.UnpivotOtherColumns(
        ConFecha, {"Time Periods","Product ID","Area/GC"}, "Tipo de Historia", "Valor"
    ),

    // 🔢 Limpieza numérica robusta (quita "," miles, respeta "." decimal)
    #"Valor a texto"           = Table.TransformColumnTypes(DatosDesplegados, {{"Valor", type text}}),
    #"Sin separador de miles"  = Table.TransformColumns(#"Valor a texto", {{"Valor", each Text.Replace(_, ",", ""), type text}}),
    #"Valor a número"          = Table.TransformColumns(#"Sin separador de miles", {{"Valor", each Number.FromText(_, "en-US"), type number}}),

    // 📊 Orden final
    ColumnasReordenadas = Table.ReorderColumns(#"Valor a número",
        {"Time Periods","Product ID","Area/GC","Tipo de Historia","Valor"})
in
    ColumnasReordenadas
